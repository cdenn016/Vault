\chapter{Geometry of Statistical Sections}
\label{ch:geometry}

A section's statistical change induces contextual geometry relative to a connection. We construct that tensor, a collective rule for choosing the connection, and conditions for their inheritance.

The role of these objects in the rest of the manuscript is limited and should be stated at the outset. No vector field, source row, channel, or record kernel in \cref{ch:dynamics} through \cref{ch:open-formation} depends on a connection, on the collective connection selected below, or on their curvature; each context evolves by a context-local flow. The formation chapters compute connections, pullback tensors, and curvature as diagnostics of those evolved sections. The separate construction in \cref{ch:contextual-coupling} adds the minimized contextual energy to the reciprocal potential and proves local well-posedness for finite graphs and a regular periodic continuum chart. The remaining extensions are recorded in \cref{open:geo-gauged-flow}.

\section{Regular law families}

\definitionheading{Fisher geometry on an admitted family}{geo:regular-family}
Let $\mathcal N$ be a finite-dimensional smooth manifold of probability laws on a standard Borel space $\mathsf Y$. In each parameter chart $\theta\mapsto P_\theta$, assume differentiability in quadratic mean with centered square-integrable score $\ell_\theta:\mathsf Y\to\R^d$. At a fixed $\theta$, this means that the Lebesgue decomposition
$P_{\theta+\delta}=p_\delta P_\theta+P_\delta^\perp$ satisfies
\begin{equation}
P_\delta^\perp(\mathsf Y)=o(|\delta|^2),
\qquad
\left\|\sqrt{p_\delta}-1-\tfrac12\delta^\top\ell_\theta\right\|_{L^2(P_\theta)}
=o(|\delta|).
\label{geo:dqm}
\end{equation}
For a tangent vector $v$, its scalar score $\ell_v$ is the corresponding directional derivative in this expansion. Require the Fisher tensor
\begin{equation}
g^F_P(v,w)=\E_P[\ell_v\ell_w]
\label{geo:fisher}
\end{equation}
to be smooth and positive definite on the admitted tangent space. Redundant parameter directions must be removed or treated as null directions before imposing this last requirement.

Condition \cref{geo:dqm} is Le Cam's differentiability in quadratic mean \citep{LeCamYang2000}. These family-specific hypotheses allow parameter-dependent supports: the laws of the family need not share null sets, and no common positive density is required \citep{AyJostLeSchwachhofer2018}. A finite-dimensional family satisfying \cref{geo:dqm} is nevertheless dominated. Differentiability in quadratic mean makes $\theta\mapsto P_\theta$ continuous in total variation, and for a countable dense set $\{\theta_n\}$ the mixture $\sum_n2^{-n}P_{\theta_n}$ dominates every $P_\theta$. The choice of the Fisher tensor is not arbitrary: up to a constant factor it is the only Riemannian metric on finite statistical simplices invariant under Markov congruent embeddings \citep{Cencov1982}, with extensions to general sample spaces under the hypotheses of \citet{AyJostLeSchwachhofer2015}. That characterization concerns the metric on one law space; it does not fix the relative weights of separate factors, which remain constitutive below.

Suppose the finite-dimensional Lie group $G$ acts smoothly on $\mathcal N$ by the sample-space pushforwards in \cref{def:agent-section}. A fixed bimeasurable bijection carries a score $\ell_v$ to $\ell_v\circ g^{-1}$. Change of variables in \cref{geo:fisher} therefore makes the action a Fisher isometry. Its infinitesimal action is
\begin{equation}
J_P\xi=\left.\frac{d}{dt}\right|_{t=0}(\exp(t\xi))_\#P,
\qquad \xi\in\mathfrak g.
\label{geo:fundamental}
\end{equation}
The regular family must be stable under every group element used in these comparisons.

\propositionheading{Fisher metric of the joint agent law}{geo:joint-fisher}
Let $\zeta_\theta(dk,dm)=q_\theta(dk\mid m)s_\theta(dm)$ be a regular family. Assume the marginal and conditional derivatives obey the product rule, that the joint score is
$\ell^\zeta_v(k,m)=\ell^s_v(m)+\ell^{q\mid m}_v(k)$, and that these scores are square-integrable. Assume differentiation of conditional normalization is valid, so
$\int\ell^{q\mid m}_v(k)q_\theta(dk\mid m)=0$ for $s_\theta$-almost every $m$. Then
\begin{equation}
g^F_\zeta(v,w)
=g^F_s(v,w)
+\int g^F_{q(\cdot\mid m)}(v,w)s(dm).
\label{geo:conditional-fisher}
\end{equation}
\begin{proof}
Expand $\E_\zeta[(\ell^s_v+\ell^{q\mid m}_v)(\ell^s_w+\ell^{q\mid m}_w)]$. Each mixed term vanishes after conditioning on $m$. The two remaining terms are exactly those displayed.
\end{proof}
This is the chain rule for Fisher information \citep{Zamir1998}, stated for a joint law with differentiable conditional normalization.

For the actual product family $q_\theta\otimes s_\theta$, \cref{geo:conditional-fisher} reduces to $g^F_q+g^F_s$. The two marginal metrics do not in general recover the joint metric. For example, the law
\begin{equation}
\zeta_\rho(k,m)=\frac{1+\rho km}{4},
\qquad k,m\in\{-1,1\},\qquad |\rho|<1,
\end{equation}
has two fixed uniform marginals, whereas its joint score is $km/(1+\rho km)$. Summing its squared score over the four outcomes gives
\begin{equation}
g^F_{\zeta_\rho}(\partial_\rho,\partial_\rho)=\frac{1}{1-\rho^2}.
\end{equation}
A declared marginal metric omits this dependence direction. We use the joint law unless a restriction is stated.

\section{Covariant statistical change}

Restrict the statistical bundle of \cref{eq:statistical-bundle} to the admitted smooth fiber $\mathcal N$, writing $E=P\times_G\mathcal N$ and $\pi:E\to\mathcal C$. A principal connection $\omega$ induces a horizontal lift $H^\omega_e:T_c\mathcal C\to T_eE$ and vertical projection $\operatorname{ver}^\omega:T_eE\to V_eE$, where $V_eE=\ker T_e\pi$ and $c=\pi(e)$. The vertical Fisher metric is well defined because the transition action is isometric.

\definitionheading{Covariant pullback}{geo:pullback}
For a smooth statistical section $\zeta:\mathcal U\to E$, define
\begin{equation}
D^\omega\zeta=\operatorname{ver}^\omega\circ T\zeta,
\qquad
h^\omega_\zeta(X,Y)
=g^F_\zeta(D^\omega\zeta X,D^\omega\zeta Y).
\label{geo:pullback-formula}
\end{equation}
The vertical projection removes the base component of $T\zeta$ forced by $\pi\circ\zeta=\id$. The resulting base tensor is positive semidefinite.

In a local frame write $A$ for the connection form and $p(c)$ for the law coordinates. With the convention that the horizontal lift is $(X,-J_pA(X))$,
\begin{equation}
D^A p(X)=dp(X)+J_pA(X).
\label{geo:local-derivative}
\end{equation}
Here $J_pA(X)$ is a statistical tangent vector.

\propositionheading{Gauge covariance and connection dependence}{geo:gauge-connection}
Under a passive change $p'=g_\#p$ and
$A'=\operatorname{Ad}_g A-(dg)g^{-1}$,
\begin{equation}
D^{A'}p'=T(g_\#)D^A p,
\qquad h^{A'}_{p'}=h^A_p.
\label{geo:passive}
\end{equation}
For two connections on the same bundle, let $\alpha=\omega'-\omega$ denote their adjoint-bundle-valued difference. Along one fixed section put $b_X=J_\zeta\alpha(X)$ and $u_X=D^\omega\zeta X$. Then
\begin{align}
D^{\omega'}\zeta X&=u_X+b_X,\\
h^{\omega'}_\zeta(X,Y)
&=h^\omega_\zeta(X,Y)+g^F(u_X,b_Y)+g^F(b_X,u_Y)+g^F(b_X,b_Y).
\label{geo:connection-change}
\end{align}
\begin{proof}
Differentiation of $p'=g_\#p$ gives $dp'=T(g_\#)dp+J_{p'}((dg)g^{-1})$. Equivariance of the fundamental action gives $J_{p'}\operatorname{Ad}_g=T(g_\#)J_p$. Substitution in \cref{geo:local-derivative} cancels the inhomogeneous terms. Fisher invariance proves \cref{geo:passive}. Replacing $A$ by $A+\alpha$ in that same derivative adds $b$; expanding the bilinear pairing gives \cref{geo:connection-change}.
\end{proof}

Changing the connection changes the comparison. \Cref{app:connection} records the transformation law of supplied connection data in a moving presentation; it does not derive a connection from the agents' laws or links.

Passive invariance does not make the tensor a property of the section alone. Suppose $G$ acts transitively on $\mathcal N$. The orbit map $g\mapsto g_\#p_0$ is then a submersion and has local sections, so a local gauge makes the section constant, $p(c)=p_0$. In that gauge \cref{geo:local-derivative} gives $D^Ap=J_{p_0}A$, and
\begin{equation}
h^A_p(X,Y)=\mathfrak k_{p_0}(A(X),A(Y)),
\qquad
\mathfrak k_{p_0}(\xi,\eta)=g^F(J_{p_0}\xi,J_{p_0}\eta).
\label{geo:transitive-pullback}
\end{equation}
As $A$ ranges over local connection forms, this ranges over every positive semidefinite form of rank at most $\dim\mathcal N$. For a homogeneous fiber the per-agent contextual tensor therefore records the supplied connection and no gauge-invariant property of the section; this is the situation of a single Gaussian family under the affine group, which acts transitively.

\propositionheading{Transported relative entropy}{geo:transported-kl}
Fix a smooth curve $\gamma$ through $c$ with $\dot\gamma(0)=X$, and transport its laws back to the fiber at $c$:
\begin{equation}
\mu_t=(T^\omega_{\gamma(t)\to c})_\#\zeta(\gamma(t)).
\end{equation}
Assume, in addition to \cref{geo:regular-family}, that $\mu_t$ has positive densities $f_t$ with respect to one measure near $t=0$, that $\log f_t$ is twice differentiable, and that both derivatives can be interchanged with the integrals of $f_t$ and $f_0\log f_t$. Assume their second derivatives are continuous at zero. Then
\begin{equation}
\KL(\zeta(c)\Vert\mu_t)
=\frac{t^2}{2}h^\omega_\zeta(X,X)+o(t^2).
\label{geo:kl-pullback}
\end{equation}
\begin{proof}
The derivative of the transported law is $D^\omega\zeta X$ by the horizontal-lift convention. Write $\ell=\partial_t\log f_t|_0$ for its score. Differentiating normalization once gives $\E_{\mu_0}\ell=0$. Differentiating twice gives
$\E_{\mu_0}[\partial_t^2\log f_t|_0+\ell^2]=0$. The first derivative of the displayed relative entropy is therefore zero and its second derivative is $\E_{\mu_0}\ell^2=h^\omega_\zeta(X,X)$. Taylor's theorem proves the claim.
\end{proof}

This is the classical second-order relation between relative entropy and Fisher information \citep{Kullback1959}, applied to the transported law. Quadratic-mean differentiability alone does not justify this second-order KL expansion.

\section{Null directions and quotients}

\propositionheading{Rank and quotient geometry}{geo:quotient}
For the positive fiber Fisher metric in \cref{geo:regular-family},
\begin{equation}
\operatorname{rad}h^\omega_\zeta=\ker D^\omega\zeta,
\qquad
\operatorname{rank}h^\omega_\zeta=\operatorname{rank}D^\omega\zeta.
\label{geo:rank}
\end{equation}
If the latter rank is constant, $K=\ker D^\omega\zeta$ is a smooth subbundle and
$T\mathcal U/K$ carries the positive-definite metric
$\bar h([X],[Y])=h^\omega_\zeta(X,Y)$.
\begin{proof}
The diagonal value of $h^\omega_\zeta$ is the squared Fisher norm of $D^\omega\zeta X$, so it vanishes exactly on the derivative's kernel. For a positive semidefinite form, its diagonal null set is its radical. Rank-nullity gives the second identity, and the constant-rank theorem for bundle maps makes $K$ a smooth subbundle. The pairing is independent of representatives because it annihilates $K$, and it is positive on every nonzero quotient class.
\end{proof}

A quotient vector bundle is not automatically the tangent bundle of a quotient manifold. Suppose a smooth Hausdorff manifold $\bar{\mathcal U}$ and a surjective submersion $q:\mathcal U\to\bar{\mathcal U}$ with connected fibers satisfy $\ker Tq=K$. The tensor descends to $\bar{\mathcal U}$ exactly when it is basic: besides annihilating $K$, it obeys $\mathcal L_Zh^\omega_\zeta=0$ for every vertical field $Z\in\Gamma(K)$. Necessity follows by differentiating a pulled-back tensor along a fiber. For sufficiency, use submersion coordinates $(y,z)$: the annihilation condition leaves only $dy^a dy^b$ components, and the Lie derivatives in the $z$ directions make their coefficients independent of $z$. Connected fibers make these local definitions agree across each fiber. The resulting tensor is positive definite by \cref{geo:rank}. A basic semidefinite tensor of this kind is a bundle-like transverse metric for the foliation by the fibers \citep{Reinhart1959,Molino1988}.

Integrability of the radical and basicness are both measured by the curvature. In a local frame write $V=D^Ap$, $j(p,\xi)=J_p\xi$, $d_pj$ for its derivative in $p$ in a chart of $\mathcal N$, and $F=dA+[A\wedge A]$ with $[A\wedge A](X,Y)=[A(X),A(Y)]$.

\propositionheading{Curvature criteria for the radical}{geo:radical-curvature}
For base vector fields $X,Y$,
\begin{equation}
V([X,Y])=\bigl(X(VY)+d_pj(VY,A(X))\bigr)-\bigl(Y(VX)+d_pj(VX,A(Y))\bigr)-J_pF(X,Y).
\label{geo:structure-identity}
\end{equation}
Consequently, where the rank of $V$ is constant, the radical $K=\ker V$ is involutive exactly when $J_pF(X,Y)=0$ for all $X,Y\in K$. For $Z\in\Gamma(K)$ and arbitrary $X,Y$,
\begin{equation}
(\mathcal L_Zh^A_p)(X,Y)=g^F\bigl(J_pF(Z,X),VY\bigr)+g^F\bigl(VX,J_pF(Z,Y)\bigr),
\label{geo:basic-criterion}
\end{equation}
so an involutive radical carries a basic tensor exactly when the right side vanishes. Both conditions hold for every flat connection.
\begin{proof}
In chart coordinates, $dp([X,Y])=X(dp(Y))-Y(dp(X))$ and $A([X,Y])=X(A(Y))-Y(A(X))-dA(X,Y)$. Differentiating $VY=dp(Y)+J_pA(Y)$ along $X$ gives $X(VY)=X(dp(Y))+d_pj(dp(X),A(Y))+J_pX(A(Y))$. Combining these identities and substituting $dp=V-J_pA$ leaves the terms of \cref{geo:structure-identity} together with $d_pj(J_pA(X),A(Y))-d_pj(J_pA(Y),A(X))$. That expression is the bracket of the fundamental fields of $A(X)$ and $A(Y)$, which for a left action equals $-J_p[A(X),A(Y)]$; with the $-J_pdA(X,Y)$ term it gives $-J_pF(X,Y)$. If $X,Y\in\Gamma(K)$, then $VX=VY=0$ identically, and \cref{geo:structure-identity} reduces to $V([X,Y])=-J_pF(X,Y)$, which proves the involutivity criterion.

For \cref{geo:basic-criterion}, write $(\mathcal L_Zh)(X,Y)=Z(g_p(VX,VY))-h([Z,X],Y)-h(X,[Z,Y])$, expand the first term with $dp(Z)=-J_pA(Z)$, and insert \cref{geo:structure-identity} with $VZ=0$. The terms containing $\xi=A(Z)$ combine into $-(\mathcal L_{\xi^*}g^F)(VX,VY)$, the Lie derivative of the Fisher metric along the fundamental field of $\xi$, which vanishes because the action is isometric. The curvature terms remain. If $F=0$, both right sides vanish.
\end{proof}

The criteria are sharp. On the trivial translation bundle over $\R^3$, use the unit-variance normal location family, the constant section of mean zero, and $A=dz-xdy$. Its location Fisher metric is one, so
\begin{equation}
D^A p=dz-xdy,\qquad h^A_p=(dz-xdy)^2.
\end{equation}
This has constant rank one, but its null fields $\partial_x$ and $\partial_y+x\partial_z$ have bracket $\partial_z$, which is not null. Here $F=-dx\wedge dy$ and $V([\partial_x,\partial_y+x\partial_z])=1=-F(\partial_x,\partial_y+x\partial_z)$, as \cref{geo:structure-identity} requires. The connection in this example is supplied, not selected: for a single location family under translations the minimizing connection of \cref{geo:collective} makes the tensor vanish, and \cref{geo:shape-pullback} shows that the minimal tensor always has an involutive radical. Gaussian laws are not required for the preceding results.

A contextual tangent quotient differs from a group-orbit quotient or measurable parent retention, which may discard information having no Fisher tangent.

\section{A collective choice of connection}

Several law sections can determine a common comparison rule by minimizing their shared frame motion. Choose smooth comparisons $\mathsf S_i$ placing the participating laws in compatible associated fibers, and write $p_i=(\mathsf S_i)_\#\zeta_i$. For bimeasurable comparisons, independent source changes $g_i$ and a common target change $g$ act by
\begin{equation}
\zeta_i'=(g_i)_\#\zeta_i,\qquad
\mathsf S_i'=g\mathsf S_i g_i^{-1},\qquad
p_i'=g_\#p_i.
\label{geo:comparison-data}
\end{equation}
The compositions in this formula mean composition of represented sample maps. More general comparisons may be declared, but their target laws must still form an admitted smooth equivariant family.

\definitionheading{Collective Fisher data}{geo:collective-data}
Let $\boldsymbol p=(p_i)_i$ lie in a finite product of regular statistical manifolds with diagonal $G$-action. Choose positive constants $w_i$ and define
\begin{align}
\mathsf G_{\boldsymbol p}(v,w)&=\sum_i w_i g^F_{p_i}(v_i,w_i),\\
J_{\boldsymbol p}\xi&=(J_{p_i}\xi)_i,\qquad
\mathfrak h_{\boldsymbol p}=\ker J_{\boldsymbol p},\qquad
H_{\boldsymbol p}=(\operatorname{im}J_{\boldsymbol p})^{\perp_{\mathsf G}},\\
\mathfrak k_{\boldsymbol p}(\xi,\eta)
&=\mathsf G_{\boldsymbol p}(J_{\boldsymbol p}\xi,J_{\boldsymbol p}\eta).
\label{geo:locked-inertia}
\end{align}
The weights and comparisons are constitutive data. A different declared joint statistical family may replace the product if its metric and group action satisfy the same hypotheses.

\propositionheading{Collective minimizing connection class}{geo:collective}
Suppose $J_{\boldsymbol p(c)}$ has constant rank on $\mathcal U$. For each contextual direction $X$, the minimizers of
\begin{equation}
a\longmapsto\|d\boldsymbol p(X)+J_{\boldsymbol p}a\|_{\mathsf G}^2
\end{equation}
form an affine class $A^F(X)+\mathfrak h_{\boldsymbol p}$, characterized by
\begin{align}
\mathfrak k(A^F(X),\xi)
&=-\mathsf G(d\boldsymbol p(X),J\xi),\\
r_X:=d\boldsymbol p(X)+JA^F(X)
&=P_Hd\boldsymbol p(X).
\label{geo:normal-equation}
\end{align}
Smooth local representatives can be chosen, linear in $X$. With $h^{\min}(X,Y)=\mathsf G(r_X,r_Y)$, any other local connection form $A$ satisfies
\begin{equation}
h^A(X,Y)
=h^{\min}(X,Y)
+\mathfrak k((A-A^F)(X),(A-A^F)(Y)).
\label{geo:minimal-pullback}
\end{equation}
The minimizing class transforms by the affine connection law, and $r$ transforms by the tangent action.
\begin{proof}
Project $-d\boldsymbol p(X)$ orthogonally onto $\operatorname{im}J$. Its preimages under $J$ give all minimizing $a$, and two preimages differ exactly by $\ker J$. Orthogonality to every $J\xi$ gives the normal equation. At constant rank, choose a smooth local complement to $\ker J$; the positive form $\mathfrak k$ restricted to it has a smooth inverse, providing the stated representative. Writing $d\boldsymbol p+JA=r+J(A-A^F)$ and using orthogonality gives \cref{geo:minimal-pullback}. Under a passive target change, the derivative and connection terms cancel exactly as in \cref{geo:passive}. Isometry of the action carries minimizers to minimizers and residuals to residuals.
\end{proof}

If infinitesimal isotropy is nonzero, the construction leaves stabilizer-valued components undetermined. An arbitrarily chosen pseudoinverse representative need not obey the affine law. Global minimizing representatives nevertheless exist on any paracompact domain of constant rank. Being minimizing at $c$ is a frame-independent property of a connection, and at each $c$ the minimizing values form an affine subspace. Given local minimizing connections $\omega_\alpha$ and a subordinate partition of unity $\phi_\alpha$, the connection $\sum_\alpha\phi_\alpha\omega_\alpha$ is an affine combination of minimizers at every point and is therefore minimizing. What requires $\ker J=0$ is uniqueness: then the local forms are unique, obey the affine law, and agree on overlaps; discrete stabilizers do not obstruct this. Curvature statements about a class should be made for $J_{\boldsymbol p}F$ on the radical, which by \cref{geo:structure-identity} is the same for every representative, or after a representative has been fixed, since the full curvature depends on that choice.

\propositionheading{Shape-space form of the minimal tensor}{geo:shape-pullback}
Assume the hypotheses of \cref{geo:collective} with $\ker J_{\boldsymbol p(c)}=0$ on $\mathcal U$. Let $\theta(v)=\mathfrak k^{-1}\mathsf G(v,J\cdot)$ denote the mechanical connection of the diagonal action, defined where the action is locally free. Then $A^F=-\boldsymbol p^*\theta$ and $F^{A^F}=-\boldsymbol p^*\Omega$, where $\Omega=d\theta-[\theta\wedge\theta]$ is horizontal. Hence $J_{\boldsymbol p}F^{A^F}(Z,\cdot)=0$ for $Z$ in the radical of $h^{\min}$; where $h^{\min}$ has constant rank, its radical is involutive and $h^{\min}$ is basic. If the action is free and proper, so that $\pi:\mathcal N^n\to\mathcal N^n/G$ is a Riemannian submersion for the quotient metric $g_{\mathrm{sh}}$, then
\begin{equation}
h^{\min}=(\pi\circ\boldsymbol p)^*g_{\mathrm{sh}},
\qquad
\operatorname{rad}h^{\min}=\ker d(\pi\circ\boldsymbol p).
\label{geo:shape-metric}
\end{equation}
\begin{proof}
The normal equation \cref{geo:normal-equation} reads $A^F(X)=-\theta(d\boldsymbol p(X))$. Pullback commutes with $d$ and with the bracket of Lie-algebra-valued forms, so $F^{A^F}=-\boldsymbol p^*(d\theta-[\theta\wedge\theta])$. For a left action $[\xi^*,\eta^*]=-[\xi,\eta]^*$, and $\theta(\xi^*)=\xi$ is constant, so $\Omega(\xi^*,\eta^*)=-\theta([\xi^*,\eta^*])-[\xi,\eta]=0$. For a horizontal field $v$, $\Omega(\xi^*,v)=-\theta([\xi^*,v])=(\mathcal L_{\xi^*}\theta)(v)$, and equivariance of $\theta$ makes this Lie derivative a commutator of $\xi$ with $\theta(v)=0$, so it vanishes. Thus $\Omega$ is horizontal. If $Z$ lies in the radical, then $r_Z=0$ and $d\boldsymbol p(Z)=-JA^F(Z)$ is an orbit direction, so $F^{A^F}(Z,\cdot)=0$; \cref{geo:radical-curvature} gives involutivity and basicness. Under a free proper action, $d\pi$ restricted to $H$ is an isometry onto the tangent space of the orbit space, and $d\pi$ annihilates $\operatorname{im}J$. Since $r_X=P_Hd\boldsymbol p(X)$, $h^{\min}(X,Y)=g_{\mathrm{sh}}(d\pi(d\boldsymbol p(X)),d\pi(d\boldsymbol p(Y)))$.
\end{proof}

The collective construction is therefore classical. The minimizing connection is the mechanical connection of \citet{Smale1970}, the orthogonal separation of rotations from shape change studied for molecules by \citet{Guichardet1984} and as a gauge field of the $n$-body problem by \citet{LittlejohnReinsch1997}; \cref{geo:minimal-pullback} is the split of kinetic energy into shape and locked-rotation parts, and $h^{\min}$ is the pullback of the quotient metric of the Riemannian submersion onto shape space \citep{ONeill1966}. What is new here is the use of the Fisher metric of the declared collective family as the kinetic metric. Rank is bounded by the shape dimension, $\operatorname{rank}h^{\min}\le\dim H=\dim\mathcal N^n-\dim(\text{orbit})$. A single transitive family has $H=0$ and zero minimizing residual; a collective family need not.

The constant-rank hypothesis fails where infinitesimal isotropy increases. Agreement need not cause such an increase: any number of unit-covariance planar normal laws with common nonzero mean $e_1$ has a free diagonal $SO(2)$ action and positive locked inertia. If the infinitesimal stabilizer does increase, the rank of $J$ drops; a smooth extension of the minimizing Lie-algebra-valued representative is then not guaranteed \citep{LewisNigamOlver2005}. For example, let $SO(2)$ act on the unit-covariance normal family on $\R^2$ and take the section $\mu(c)=(\epsilon,c)$ for one agent. Then $J\xi=\xi(-c,\epsilon)$, $\mathfrak k=c^2+\epsilon^2$, and
\begin{equation}
A^F(\partial_c)=-\frac{\epsilon}{c^2+\epsilon^2},
\qquad
\int_{\R}A^F(\partial_c)dc=-\pi,
\end{equation}
so as $\epsilon\to0$ the connection concentrates to $-\pi\delta(c)$ at the isotropic point. The minimal tensor stays bounded, $h^{\min}(X,X)\le\Vert d\boldsymbol p(X)\Vert_{\mathsf G}^2$ by the choice $a=0$. This one-dimensional example exhibits connection concentration, not curvature blowup. A smooth collective connection and its curvature require the relevant regularity hypotheses; the displayed bound for $h^{\min}$ alone does not control derivatives of a connection near a change of orbit type.

The constant weights $w_i$ in \cref{geo:collective-data} have a probabilistic reading. With $\sum_iw_i=1$, the weighted metric is the Fisher metric, in the $\boldsymbol p$ directions, of the labeled family $\sum_iw_i\delta_i\otimes p_i$. If the weights depend on the context, its Fisher metric acquires the term $\sum_idw_i\otimes dw_i/w_i$. The group acts on samples and not on labels, so the weight scores have zero Fisher pairing with every orbit direction; the normal equation is unchanged and $h^{\min}$ gains that connection-independent term. A correlated population law, whose Fisher metric sees dependence directions invisible to every marginal (\cref{geo:joint-fisher}), is not of this weighted product form and is not treated here.

The collective connection is also the auxiliary field of a gauge-invariant contextual energy. For a Riemannian metric $\gamma$ on a domain of $\mathcal C$ with volume form $\mathrm{dvol}_\gamma$, put
\begin{equation}
\mathcal E_{\mathrm{ctx}}[\boldsymbol p,A]
=\tfrac12\int\gamma^{\mu\nu}\mathsf G\bigl(D^A_\mu\boldsymbol p,D^A_\nu\boldsymbol p\bigr)\mathrm{dvol}_\gamma.
\label{geo:context-energy}
\end{equation}
By \cref{geo:passive} it is gauge invariant. Its integrand is a sum over directions of the costs minimized in \cref{geo:collective}, so at fixed $\boldsymbol p$ its minimizers over $A$ are the minimizing connections, and the minimum is $\tfrac12\int\Tr_\gamma h^{\min}\mathrm{dvol}_\gamma$. This is the structure of the $\mathbb{CP}^{N-1}$ sigma model, whose gauge field has no kinetic term and is eliminated algebraically \citep{DAddaLuscherDiVecchia1978}; the analogy is structural and does not import that model's quantum dynamics.

\openproblemheading{Extensions of the contextual Fisher flow}{open:geo-gauged-flow}
\Cref{ch:contextual-coupling} supplies a finite-graph construction and a continuum local theorem on a flat torus. Auxiliary elimination diffuses the law shape and leaves frame and source-row equations without independent diffusion. Global regularity, changes of orbit type, physical boundaries, independently evolving rows and communication links, and a graph-refinement theorem remain open. The quotient target can be open and positively curved, so the local theorem alone gives no global continuation. Deriving a kinetic term for the connection from a declared ensemble of contextual sections is a separate problem; that ensemble is not supplied.

\propositionheading{Aggregate covariant derivative}{geo:aggregate}
Let $\mathcal M$ be a smooth equivariant map from the collective statistical family to a finite-dimensional representation $W$ of $G$. For $M=\mathcal M(\boldsymbol p)$ define
\begin{equation}
e(X)=d\mathcal M_{\boldsymbol p}r_X
=dM(X)+\rho_*(A^F(X))M.
\label{geo:aggregate-jet}
\end{equation}
This $W$-valued one-form is independent of the representative of the minimizing class. If $\dim\mathcal C=\dim W$, it is an isomorphism exactly when the actual map $d\mathcal M_{\boldsymbol p}\circ r$ has full rank.
\begin{proof}
Differentiating equivariance gives $d\mathcal M_{\boldsymbol p}J\xi=\rho_*(\xi)M$. Applying this identity to \cref{geo:normal-equation} proves \cref{geo:aggregate-jet}. A change in $A^F$ belonging to $\ker J$ has zero image under $d\mathcal M$, proving representative independence. The rank statement is the criterion for a linear map between equal-dimensional spaces to be invertible.
\end{proof}

Surjectivity of $d\mathcal M|_H$ establishes that a full-rank first jet can be chosen at the law tuple: choose horizontal preimages of a basis of $W$ and realize those as derivatives in a local statistical chart. It does not establish full rank for the supplied section or its preservation under inference. The determinant specialization in \cref{app:determinant} uses an additional form on $W$; the Fisher metric above remains positive. Surjectivity itself is a restriction on the number and configuration of the agents; \cref{geo:det-rank} computes it for proper complex Gaussian targets.

\section{Information retained by a channel}

\theoremheading{Score projection and exact Fisher loss}{geo:fisher-loss}
Let $\{P_\theta\}$ satisfy \cref{geo:dqm} at $\theta_0$, and let $K:\mathsf Y\rightsquigarrow\bar{\mathsf Y}$ be one normalized Markov kernel independent of $\theta$. Under the joint law $P_{\theta_0}(dy)K(y,d\bar y)$, the output family is differentiable in quadratic mean with score and Fisher matrix
\begin{align}
\bar\ell(\bar Y)&=\E[\ell(Y)\mid\bar Y],\\
I(P_{\theta_0})-I(P_{\theta_0}K)
&=\E\operatorname{Cov}(\ell(Y)\mid\bar Y)\succeq0.
\label{geo:score-projection}
\end{align}
Equality in direction $v$ holds exactly when $v^\top\ell(Y)$ is measurable from $\bar Y$ up to a joint null set.
\begin{proof}
Lift the decomposition in \cref{geo:dqm} to the joint space. Its absolutely continuous square root is $V_\delta=\sqrt{p_\delta(Y)}$ with the same $L^2$ remainder, and its singular mass is still $o(|\delta|^2)$. For its output, put
$m_\delta=\E[V_\delta\mid\bar Y]$ and
$v_\delta=\E[(V_\delta-m_\delta)^2\mid\bar Y]$. The projected density of this absolutely continuous part is
$q_\delta=m_\delta^2+v_\delta$.
Conditional expectation is an $L^2$ contraction, so
$m_\delta=1+\tfrac12\delta^\top\E[\ell\mid\bar Y]+o_{L^2}(|\delta|)$.

To replace $m_\delta$ by $\sqrt{q_\delta}$, observe that $m_\delta\ge0$ and
\begin{equation}
(\sqrt{q_\delta}-m_\delta)^2
=v_\delta
\frac{v_\delta}{(\sqrt{q_\delta}+m_\delta)^2}
\le v_\delta.
\end{equation}
Set the fraction to zero where its numerator and denominator both vanish. It tends to zero in probability. Moreover $v_\delta/|\delta|^2$ is uniformly integrable as $\delta\to0$: the square-root expansion expresses
$(V_\delta-1)/|\delta|$ as a bounded finite-dimensional combination of the fixed $L^2$ scores plus an $L^2$ remainder tending to zero, and conditional variance preserves the resulting uniform integrability. Thus the last display has expectation $o(|\delta|^2)$.
The projected singular part may acquire an absolutely continuous density $a_\delta$, but its total mass is $o(|\delta|^2)$. The inequality
$(\sqrt{q_\delta+a_\delta}-\sqrt{q_\delta})^2\le a_\delta$
shows that this change does not affect the score; any remaining singular output mass has the same small bound. This proves output differentiability and its score formula. Total covariance gives the matrix identity. Its quadratic form in $v$ is the expected conditional variance of $v^\top\ell$, which vanishes exactly under the stated measurability condition.
\end{proof}

This score-projection mechanism is the one in \citet[Theorem~7]{Pollard2013}; the Markov formulation is also treated in \citet{AyJostLeSchwachhofer2018}. It identifies local score sufficiency. It does not identify a kernel recovering every law in the experiment; that an insufficient statistic can preserve Fisher information is the phenomenon of \citet{KaganShepp2005}. For example, take independent Bernoulli variables with probabilities $1/2+\theta/4$ and $1/2+\theta^2/4$, and discard the second. At $\theta=0$ its score vanishes, so no Fisher information is lost. A fixed recovery kernel applied to the first Bernoulli law makes every reconstructed event probability affine in $\theta$, whereas the required second-coordinate probability is quadratic. Such a kernel cannot recover the family on a neighborhood.

\propositionheading{Moving-channel correction}{geo:moving-channel}
Let the channel instead have density $k_\theta(\bar y\mid y)$ with respect to a fixed output measure. Assume the joint family is differentiable in quadratic mean with square-integrable score
$\ell_\theta(Y)+b_\theta(Y,\bar Y)$, where
$b_\theta=\partial_\theta\log k_\theta$, and assume differentiation of channel normalization gives $\E[b_\theta\mid Y]=0$. Then
\begin{align}
\bar\ell_\theta(\bar Y)
&=\E_\theta[\ell_\theta(Y)+b_\theta(Y,\bar Y)\mid\bar Y],\\
I(\bar Y)
&=I(Y)+\E_\theta I(K_\theta(Y,\cdot))
-\E_\theta\operatorname{Cov}(\ell_\theta+b_\theta\mid\bar Y).
\label{geo:moving-score}
\end{align}
\begin{proof}
Apply \cref{geo:fisher-loss} to the fixed output projection of the joint experiment. Its score is the stated sum. Conditional centering of $b_\theta$ eliminates the mixed terms in its Fisher matrix, leaving the input matrix plus the average conditional channel matrix. Subtracting the conditional covariance proves the formula.
\end{proof}

Consequently input-to-output Fisher change has no fixed sign for a moving channel. A constant input followed by a Bernoulli output with success probability $(1+e^{-\theta})^{-1}$ has zero input Fisher information and output information $e^{-\theta}/(1+e^{-\theta})^2>0$. A singular moving deterministic statistic need not have a channel log density at all. Its actual weak derivative, schematically $d(PK)=(dP)K+P(dK)$ when these derivatives exist, must replace that expression. These distinctions apply to the law-dependent channels used in the interaction potential.

\section{Geometry across a retention}

Let $\Psi:E\to\bar E$ be a smooth bundle map covering $f:\mathcal C\to\bar{\mathcal C}$, where both statistical bundles carry positive vertical Fisher metrics and connections $\omega,\bar\omega$. Assume the sections are related,
\begin{equation}
\Psi\circ\zeta=\bar\zeta\circ f.
\label{geo:related-sections}
\end{equation}
This is a hypothesis: a map over $f$ does not automatically make every fine section descend through $f$.

Such maps arise from equivariant kernels. Let $\tilde f:P\to\bar P$ be a principal bundle map covering $f$ with homomorphism $\phi:G\to\bar G$, and let $\kappa:\mathsf Y\rightsquigarrow\bar{\mathsf Y}$ be a Markov kernel with $\kappa(gy,\cdot)=\phi(g)_\#\kappa(y,\cdot)$. Then $\Psi[u,\zeta]=[\tilde f(u),\zeta\kappa]$ is well defined, since $[ug,\zeta]$ and $[u,g_\#\zeta]$ are sent to $[\tilde f(u)\phi(g),\zeta\kappa]$ and $[\tilde f(u),\phi(g)_\#(\zeta\kappa)]$, which coincide. It is fiberwise affine, contracts relative entropy by data processing, and for a fixed kernel gives $\Delta_F\succeq0$ below by \cref{geo:fisher-loss}.

\definitionheading{Horizontal transport defect}{geo:horizontal-defect}
For $X\in T_c\mathcal C$, define
\begin{equation}
a_\Psi(X)=T\Psi(H^\omega_{\zeta(c)}X)
-H^{\bar\omega}_{\Psi(\zeta(c))}(TfX).
\label{geo:horizontal-defect-formula}
\end{equation}
Both terms project to $TfX$, so their difference lies in the same coarse vertical tangent space. Set
$u_X=D^\omega\zeta X$, $\Psi_*=T^V\Psi$, and
$\Delta_F=g^F-\Psi_*^*\bar g^F$ along the section.

\propositionheading{Signed comparison of base tensors}{geo:signed-comparison}
Under \cref{geo:related-sections},
\begin{align}
D^{\bar\omega}\bar\zeta(TfX)&=\Psi_*u_X+a_\Psi(X),\\
h^\omega_\zeta-f^*h^{\bar\omega}_{\bar\zeta}
&=u^*\Delta_F-\mathcal X_\Psi-\mathcal Q_\Psi,
\label{geo:signed-identity}
\end{align}
where
\begin{align}
\mathcal X_\Psi(X,Y)
&=\bar g^F(\Psi_*u_X,a_\Psi(Y))
+\bar g^F(a_\Psi(X),\Psi_*u_Y),\\
\mathcal Q_\Psi(X,Y)&=\bar g^F(a_\Psi(X),a_\Psi(Y)).
\end{align}
If each fiber map is realized by a fixed parameter-independent Markov kernel, then $\Delta_F\succeq0$. In that case base contraction holds exactly when, for every $X$,
\begin{equation}
2\bar g^F(\Psi_*u_X,a_\Psi(X))
+\|a_\Psi(X)\|_{\bar g^F}^2
\le\Delta_F(u_X,u_X).
\label{geo:contraction-criterion}
\end{equation}
\begin{proof}
Differentiate \cref{geo:related-sections}, decompose $T\zeta X=H^\omega X+u_X$, and take coarse vertical parts. This proves the first identity. Expand the coarse Fisher pairing of $\Psi_*u+a_\Psi$ and subtract it from the fine pairing to obtain the second. For a fixed Markov fiber map, \cref{geo:fisher-loss} gives $\Delta_F\succeq0$. Evaluating the signed identity on the diagonal proves the exact contraction criterion.
\end{proof}

Zero horizontal defect is sufficient, but is not necessary, for \cref{geo:contraction-criterion}. For a witness, take the unit-variance normal location family under translations on both levels, a fiber map adding independent unit-variance normal noise, so that the coarse location Fisher metric is $\tfrac12$ and $\Delta_F=\tfrac12$, and connections differing by a constant $c$, so that $a_\Psi=c$ in the location coordinate. For $u_X=1$, the base tensors are $1$ and $\tfrac12(1+c)^2$. The difference $1-\tfrac12(1+c)^2$ equals $\tfrac{79}{200}$ at $c=\tfrac1{10}$, which is contraction with a nonzero defect, and $-\tfrac18$ at $c=\tfrac12$, where contraction fails. Information discarded in the fiber and mismatch of horizontal transports enter different terms. For two successive smooth statistical maps $R_{01},R_{12}$, the vertical defects also compose exactly:
\begin{equation}
\Delta_{02}=\Delta_{01}+R_{01}^*\Delta_{12},
\qquad
\Delta_{ab}=g_a-R_{ab}^*g_b.
\label{geo:defect-composition}
\end{equation}
This follows by inserting and subtracting $R_{01}^*g_1$ and using contravariance of pullback. Both terms are nonnegative when the two maps are fixed Markov morphisms.

\propositionheading{Retention of the collective connection}{geo:retained-connection}
Let $R:\mathcal N\to\bar{\mathcal N}$ be a smooth $G$-equivariant map between positive-metric collective statistical families, independent of the contextual coordinate. Denote their metrics by $g,\bar g$, their infinitesimal orbit maps by $J,\bar J$, and put $\Delta=g-R^*\bar g$. Let $\boldsymbol p(c)$ be a smooth fine section, and let $A^F$ and $r_X=d\boldsymbol p(X)+JA^F(X)$ be a fine minimizing representative and residual. The same representative minimizes the cost at $\bar{\boldsymbol p}=R(\boldsymbol p)$ exactly when
\begin{equation}
\Delta(r_X,J\xi)=0
\quad\text{for all }X,\xi.
\label{geo:connection-retention}
\end{equation}
Under this condition, equality of the entire fine and coarse affine minimizing classes holds exactly when $\ker J=\ker\bar J$. For every possible fine first jet, the condition is $\Delta(H,\operatorname{im}J)=0$. If $\Delta\succeq0$, vanishing loss on every orbit direction is sufficient. This includes a fixed Markov morphism equipped with the Fisher metrics of its input and output experiments in the same normalization; arbitrary independently reweighted metrics need not satisfy that premise.
\begin{proof}
Equivariance gives $dR J=\bar J$. Thus the coarse residual using $A^F$ is $dR r_X$. Its normal equation is $\bar g(dR r_X,\bar J\xi)=0$. Fine orthogonality gives $g(r_X,J\xi)=0$, so the coarse equation is equivalent to \cref{geo:connection-retention}. Once the representative minimizes both costs, their classes are $A^F+\ker J$ and $A^F+\ker\bar J$, proving the equality criterion. Every horizontal vector is realizable as a residual by choosing the first jet equal to that vector, which gives the all-jets statement. When $\Delta\succeq0$, Cauchy--Schwarz for this semidefinite form makes $\Delta(v,J\xi)=0$ whenever $\Delta(J\xi,J\xi)=0$.
\end{proof}

Contraction alone is insufficient. Two independent unit-variance normal location laws with $J\xi=(\xi,\xi)$ and means $(x,-x)$ have Fisher matrix $I$ and minimizing coefficient zero. Adding independent unit normal noise only to the first sample is a fixed equivariant channel. Its coarse Fisher matrix is $\operatorname{diag}(1/2,1)$, whose cost $\tfrac12(1+a)^2+(-1+a)^2$ is minimized at $a=1/3$. The minimizing connection has changed despite Fisher contraction. Its preservation is an additional requirement beyond the dynamical and response conditions in \cref{ch:inheritance}. The minimal tensor itself nevertheless contracts, and the connection shift is measured exactly.

\corollaryheading{Exact contraction of the minimal tensor}{geo:minimal-contraction}
In the setting of \cref{geo:retained-connection}, assume that the coarse orbit map $\bar J$ also has constant rank along $\bar{\boldsymbol p}=R(\boldsymbol p)$, and let $\bar A^F$ be a coarse minimizing representative with minimal tensor $\bar h^{\min}$ and locked inertia $\bar{\mathfrak k}$. Then
\begin{equation}
h^{\min}(X,X)-\bar h^{\min}(X,X)
=\Delta(r_X,r_X)
+\bar{\mathfrak k}\bigl((A^F-\bar A^F)(X),(A^F-\bar A^F)(X)\bigr).
\label{geo:minimal-contraction-identity}
\end{equation}
The right side does not depend on the coarse representative. If $\Delta\succeq0$, in particular for a fixed Markov morphism with Fisher metrics in the same normalization, both terms are nonnegative, so $\bar h^{\min}\le h^{\min}$; the second term vanishes exactly when \cref{geo:connection-retention} holds.
\begin{proof}
By equivariance, the coarse residual of the fine representative is $dR(r_X)$. Applying \cref{geo:minimal-pullback} at the coarse level to the connection $A^F$ gives $\bar g(dR(r_X),dR(r_X))=\bar h^{\min}(X,X)+\bar{\mathfrak k}((A^F-\bar A^F)(X),(A^F-\bar A^F)(X))$. The left side equals $g(r_X,r_X)-\Delta(r_X,r_X)=h^{\min}(X,X)-\Delta(r_X,r_X)$. The form $\bar{\mathfrak k}$ annihilates $\ker\bar J$, which gives representative independence. The last statement is the criterion of \cref{geo:retained-connection}.
\end{proof}
In the two-location example above, $h^{\min}=2$, $\bar h^{\min}=\tfrac43$, $\Delta(r,r)=\tfrac12$, and the inertia term is $\tfrac32(\tfrac13)^2=\tfrac16$. For successive equivariant retentions the identity telescopes, each step contributing its own pair of nonnegative terms. Combined with \cref{geo:shape-pullback}, when $\Delta\succeq0$ the map induced by $R$ between shape spaces does not increase the quotient metric along the section.
