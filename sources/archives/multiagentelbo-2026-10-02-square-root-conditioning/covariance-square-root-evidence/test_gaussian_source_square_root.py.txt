"""Tiny numerical grounding for the equivalent hidden-factor source update."""
from __future__ import annotations

import importlib

import pytest
import torch

from test_gaussian_source_gaussian import _example, _projection


def test_factor_update_retains_small_positive_local_variance() -> None:
    api = importlib.import_module("multiagent_elbo.realizations.gaussian.source_square_root")
    Sigma = torch.tensor([[[1., .5], [.5, 1.]]], dtype=torch.float64)
    factor = torch.linalg.cholesky(Sigma)
    local_rows = torch.tensor([[1., 0.]], dtype=factor.dtype)
    shift = torch.zeros((1, 1), dtype=factor.dtype)
    tilted_factor = torch.full((1, 1, 1), 2.**-30, dtype=factor.dtype)
    displacement, updated = api.condition_hidden_factor(factor, local_rows, shift, tilted_factor)
    covariance = updated@updated.mT
    torch.testing.assert_close(displacement, torch.zeros((1, 2), dtype=factor.dtype), atol=0., rtol=0.)
    torch.testing.assert_close(covariance[:, 0, 0], torch.full((1,), 2.**-60, dtype=factor.dtype), atol=0., rtol=1e-13)
    torch.testing.assert_close(covariance[:, 1, 0], torch.full((1,), 2.**-61, dtype=factor.dtype), atol=0., rtol=1e-13)
    torch.linalg.cholesky(covariance)
    assert updated.shape == factor.shape


@pytest.mark.parametrize("block_size", [2, 3])
def test_square_root_rows_match_existing_dense_cubature_and_fiber(block_size: int) -> None:
    api = importlib.import_module("multiagent_elbo.realizations.gaussian.source_square_root")
    dense = importlib.import_module("multiagent_elbo.realizations.gaussian.source_gaussian")
    mu, covariance, model, state = _example(block_size=block_size)
    before_mu, before_covariance = mu.clone(), covariance.clone()
    labels = torch.tensor([[0, 1, 0, 1, 0, 1], [1, 0, 1, 0, 1, 0]])
    expected = dense.sequential_source_message(mu, covariance, labels, labels, state, model, block_size)
    actual = api.square_root_source_rows(mu, covariance, labels, state, model, block_size)
    selected = actual["log_rows"].gather(-1, labels[..., None]).squeeze(-1).sum(-1)
    torch.testing.assert_close(selected, expected["log_refresh"], atol=3e-12, rtol=3e-12)
    torch.testing.assert_close(actual["mean"], expected["mean"], atol=3e-12, rtol=3e-12)
    torch.testing.assert_close(actual["covariance"], expected["covariance"], atol=3e-12, rtol=3e-12)
    P = _projection(6, 2, block_size)
    torch.testing.assert_close(P@actual["covariance"], torch.zeros_like(P@covariance), atol=2e-13, rtol=0.)
    torch.testing.assert_close(actual["mean"].flatten(1)@P.T, mu.flatten(1)@P.T, atol=2e-13, rtol=0.)
    torch.testing.assert_close(actual["log_rows"].logsumexp(-1), torch.zeros((2, 6), dtype=mu.dtype), atol=1e-14, rtol=0.)
    torch.testing.assert_close(mu, before_mu, atol=0., rtol=0.)
    torch.testing.assert_close(covariance, before_covariance, atol=0., rtol=0.)


def test_full_retention_uses_native_rows_without_hidden_factor() -> None:
    from multiagent_elbo.realizations.gaussian.switching_paths import source_log_posterior
    api = importlib.import_module("multiagent_elbo.realizations.gaussian.source_square_root")
    mu, covariance, model, state = _example(block_size=1)
    labels = torch.zeros((2, 6), dtype=torch.long)
    actual = api.square_root_source_rows(mu, covariance, labels, state, model, 1)
    torch.testing.assert_close(actual["log_rows"], source_log_posterior(mu, state, model))
    torch.testing.assert_close(actual["mean"], mu, atol=0., rtol=0.)
    torch.testing.assert_close(actual["covariance"], covariance, atol=0., rtol=0.)
