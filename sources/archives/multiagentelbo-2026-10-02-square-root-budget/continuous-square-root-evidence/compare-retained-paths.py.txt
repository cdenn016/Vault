from pathlib import Path
import json
import numpy as np
root=Path(__file__).resolve().parents[2]
new=root/'artifacts/gaussian-continuous-square-root/20261003T041753622698Z'
old=root/'docs/research/2026-10-02-renormalization-completion/continuous-budget-evidence'
assert json.loads((new/'metadata.json').read_text())['status']=='pass'
keys=['initial_X','initial_J','proposal_times','proposal_uniforms','probe_times','probe_uniforms']
keys += [grid+'_'+key for grid in ['coarse','fine'] for key in ['native','assignments','times','dW']]
comparisons=[]
for previous in sorted(old.glob('N*-path*.npz')):
    with np.load(previous) as a, np.load(new/previous.name) as b:
        errors={key:float(np.max(np.abs(a[key]-b[key]))) if a[key].size else 0. for key in keys}
        exact={key:bool(np.array_equal(a[key],b[key])) for key in keys}
    comparisons.append({'path':previous.name,'exact':all(exact.values()),'maximum_absolute_errors':errors})
with np.load(root/'docs/research/2026-10-02-renormalization-completion/covariance-square-root-path-evidence/path-output.npz') as a, np.load(new/'N1024-path001.npz') as b:
    differences={key:float(np.max(np.abs(a[key]-b['fine_'+key]))) for key in a.files}
    exact={key:bool(np.array_equal(a[key],b['fine_'+key])) for key in a.files}
report={'scope':'Posthoc deterministic native-tape and validation-context comparisons; no study acceptance tolerance added',
        'original_study_paths_compared':len(comparisons),'all_native_tapes_exact':all(x['exact'] for x in comparisons),
        'native_comparisons':comparisons,'validation_path_fine_exact':all(exact.values()),
        'validation_path_maximum_absolute_errors':differences}
(new/'retained-path-comparison.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
print(json.dumps({k:v for k,v in report.items() if k!='native_comparisons'},indent=2))
