from pathlib import Path
import sympy as s,json,time
out=Path('artifacts/binary-reverse-budget'); start=time.perf_counter()
y,R,t,u=s.symbols('y R t u',real=True)
energy_difference=s.expand(y*y/2-(y-2*R)**2/2)
assert s.expand(energy_difference-2*R*(y-R))==0
# sigma(2t(y-t)) = exp(t(y-t))/(2 cosh(t(y-t))).
exponent=s.expand((-y*y/2+t*(y-t)).subs(y,t+u/t)+t*t/2+u*u/(2*t*t))
assert exponent==0
result={'fixed_energy_logit':str(energy_difference),'tilted_density_exponent_residual':str(exponent),'scope':'Exact scalar algebra only; asymptotic limit follows from the documented dominated-convergence proof. No trajectory or GPU.','sympy_version':s.__version__,'elapsed_seconds':time.perf_counter()-start}
(out/'native-update-symbolic.json').write_text(json.dumps(result,indent=2)+'\n'); print(json.dumps(result,indent=2))
